   func currentDayCount(keyID string) int {
       val, ok := perDayCounters.Load(keyID)
       if !ok {
           return 0
       }
       rc := val.(*rateCounter)
       rc.mu.Lock()
       defer rc.mu.Unlock()
       if rc.windowID != time.Now().UTC().YearDay() {
           return 0
       }
       return rc.count
   }

   // InjectDayCount sets a perDayCounters entry for the given keyID with today's windowID

   func InjectDayCount(keyID string, count int) {
       rc := &rateCounter{count: count, windowID: time.Now().UTC().YearDay()}
       perDayCounters.Store(keyID, rc)

   // InjectDayCountStale sets a perDayCounters entry for the given keyID with a windowID

   func InjectDayCountStale(keyID string) {

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client → GET /api/keys | Admin bearer token required; no unauthenticated access to rate limit data |
| in-memory counter → HTTP response | perDayCounters read under mutex with windowID staleness check |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-10-01 | Information Disclosure | `day_usage` field in `/api/keys` response | mitigate | Field only appears in `/api/keys` which is gated by `adminMiddleware` (Bearer admin key check via `crypto/subtle.ConstantTimeCompare`). No public exposure. |
| T-10-02 | Information Disclosure | Stale perDayCounters value revealing prior-day request count | mitigate | `currentDayCount` validates `rc.windowID == time.Now().UTC().YearDay()` before returning count; returns 0 if stale. Staleness check implemented and tested. |
| T-10-03 | Spoofing | Admin key header forgery | accept | Constant-time comparison already implemented in `adminMiddleware`. No change in this plan. |
| T-10-04 | Denial of Service | Repeated GET /api/keys for all keys triggering N counter reads | accept | Read path is O(1) per key (sync.Map.Load + mutex read). No DB call. No amplification risk at this scale. |
</threat_model>
