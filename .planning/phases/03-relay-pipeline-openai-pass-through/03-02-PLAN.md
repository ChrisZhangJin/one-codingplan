    ```go
    func (s *Server) proxyStream(c *gin.Context, resp *http.Response, keyID string, upstreamID uint, start time.Time) {
        defer resp.Body.Close()

        // Set streaming headers before any body write (D-09)
        c.Writer.Header().Set("Content-Type", "text/event-stream")
        c.Writer.Header().Set("X-Accel-Buffering", "no")       // prevent nginx buffering
        c.Writer.Header().Set("Cache-Control", "no-cache")
        c.Writer.Header().Set("Connection", "keep-alive")
        c.Writer.WriteHeader(http.StatusOK)

        flusher, ok := c.Writer.(http.Flusher)
        if !ok {
            // Should not happen with net/http, but handle gracefully
            s.logUsage(keyID, upstreamID, false, 0, 0, time.Since(start))
            return
        }

        // Start heartbeat goroutine (per CLAUDE.md tech stack: 30s ticker)
        done := make(chan struct{})
        defer close(done)
        go func() {
            ticker := time.NewTicker(heartbeatInterval)
            defer ticker.Stop()
            for {
                select {
                case <-ticker.C:
                    fmt.Fprint(c.Writer, ": heartbeat\n\n")
                    flusher.Flush()
                case <-done:
                    return
                }
            }
        }()

        // Copy upstream SSE stream to client (D-09: io.Copy + Flusher)
        // Use a small buffer to ensure frequent flushes
        buf := make([]byte, 4096)
        for {
            n, err := resp.Body.Read(buf)
            if n > 0 {
                c.Writer.Write(buf[:n])
                flusher.Flush()
            }
            if err != nil {
                break // EOF or mid-stream error (D-10: just close, no retry)
            }
        }

        // Stream complete -- log usage async (D-14)
        // Token counts = 0 for streaming per D-13 fallback
        s.logUsage(keyID, upstreamID, true, 0, 0, time.Since(start))
    }
    ```

    Key implementation details:
    - Use manual `resp.Body.Read` + `c.Writer.Write` + `flusher.Flush()` loop instead of `io.Copy` to ensure each chunk is flushed immediately (io.Copy uses 32KB default buffer which may delay small SSE frames)
    - `defer close(done)` ensures heartbeat goroutine exits on all return paths (prevents goroutine leak per Pitfall 5)
    - Mid-stream failure (D-10): when `resp.Body.Read` returns an error (not io.EOF), the function returns, closing the client connection naturally. No retry is attempted because HTTP 200 was already committed.
    - `heartbeatInterval` is a package-level var (not const) so tests can override it
    - Token counts are 0 for streaming responses per D-13 ("otherwise log 0")

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| upstream -> client (SSE stream) | Upstream response bytes copied directly to client; semi-trusted content |
| heartbeat goroutine -> client writer | Background goroutine writes to same ResponseWriter as main handler |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-3-08 | Denial of Service | proxyStream heartbeat goroutine | mitigate | `defer close(done)` ensures goroutine exits when handler returns; no goroutine leak on any code path |
| T-3-09 | Tampering | SSE frame passthrough | accept | Upstream SSE frames are passed through unmodified; ocp does not parse or validate SSE content -- upstream is semi-trusted (admin-configured) |
| T-3-10 | Denial of Service | upstream sends infinite stream | mitigate | relayClient.Timeout (30s) applies to idle reads; if upstream stops sending data, read will timeout; client can also disconnect which cancels the context |
| T-3-11 | Information Disclosure | streaming error response | mitigate | Once HTTP 200 is committed, no error body is sent; connection simply closes on failure (D-10) |
</threat_model>
