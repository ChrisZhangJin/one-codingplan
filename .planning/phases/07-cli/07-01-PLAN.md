1. Create `cmd/ocp/serve.go` in package `main`. This file contains the `serveCmd` Cobra command that wraps ALL existing server startup logic currently in main.go.

The serve command:
- Has `Use: "serve"` and `Short: "Start the ocp proxy server"`
- Defines a local `--config` flag (string, default `"./config.yaml"`) per D-03
- Its `RunE` function contains the exact logic currently in `main()`: load config, validate OCP_ENCRYPTION_KEY, open DB, migrate, sync upstreams, create pool, set model overrides and formats, start probe loop, create server, run engine
- On error, return `fmt.Errorf(...)` instead of `log.Fatalf` so Cobra handles exit

```go
package main

import (
	"fmt"
	"log"
	"os"

	"github.com/spf13/cobra"

	"one-codingplan/internal/config"
	"one-codingplan/internal/database"
	"one-codingplan/internal/pool"
	"one-codingplan/internal/server"
)

var serveConfigPath string

var serveCmd = &cobra.Command{
	Use:   "serve",
	Short: "Start the ocp proxy server",
	RunE: func(cmd *cobra.Command, args []string) error {
		cfg, err := config.Load(serveConfigPath)
		if err != nil {
			return fmt.Errorf("load config: %w", err)
		}

		encKey := []byte(os.Getenv("OCP_ENCRYPTION_KEY"))
		if len(encKey) != 16 && len(encKey) != 24 && len(encKey) != 32 {
			return fmt.Errorf("OCP_ENCRYPTION_KEY must be set to a 16, 24, or 32 byte string (got %d bytes)", len(encKey))
		}

		db, err := database.Open(cfg.Database.Path)
		if err != nil {
			return fmt.Errorf("open db: %w", err)
		}

		if err := database.Migrate(db); err != nil {
			return fmt.Errorf("migrate: %w", err)
		}

		if err := database.SyncUpstreams(db, cfg.Upstreams, encKey); err != nil {
			return fmt.Errorf("sync upstreams: %w", err)
		}

		poolCfg := &pool.Config{
			RateLimitBackoff: cfg.PoolBackoff(),
		}
		p, err := pool.New(db, encKey, poolCfg)
		if err != nil {
			return fmt.Errorf("init pool: %w", err)
		}
		for _, u := range cfg.Upstreams {
			if u.ModelOverride != "" {
				p.SetModelOverride(u.Name, u.ModelOverride)
			}
			if u.Format != "" {
				p.SetFormat(u.Name, u.Format)
			}
		}
		p.StartProbeLoop()
		defer p.Stop()

		srv := server.New(db, cfg, p)
		r := srv.Engine()

		addr := fmt.Sprintf(":%d", cfg.Server.Port)
		log.Printf("ocp starting on %s", addr)
		return r.Run(addr)
	},
}

func init() {
	serveCmd.Flags().StringVar(&serveConfigPath, "config", "./config.yaml", "path to config file")
	rootCmd.AddCommand(serveCmd)
}
```
