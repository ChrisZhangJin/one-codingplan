       ```go
       func (s *Server) handleToggleUpstream(c *gin.Context) {
           id := c.Param("id")

           var upstream models.Upstream
           if err := s.db.First(&upstream, "id = ?", id).Error; err != nil {
               c.JSON(http.StatusNotFound, gin.H{"error": "upstream not found"})
               return
           }

           newEnabled := !upstream.Enabled
           if err := s.db.Model(&upstream).Update("enabled", newEnabled).Error; err != nil {
               c.JSON(http.StatusInternalServerError, gin.H{"error": "failed to toggle upstream"})
               return
           }

           // Reload pool from database to pick up the change
           if err := s.pool.Reload(s.db); err != nil {
               // Log but don't fail — DB state is correct, pool will catch up on restart
               c.JSON(http.StatusOK, gin.H{"enabled": newEnabled, "warning": "pool reload failed"})
               return
           }

           c.JSON(http.StatusOK, gin.H{"enabled": newEnabled})
       }
       ```

|----------|-------------|
| Browser -> /api/upstreams/:id/toggle | New mutation endpoint; requires admin auth |
| Dashboard JS -> /api/* | All dashboard API calls use Bearer token from localStorage |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-6-08 | Spoofing | handleToggleUpstream | mitigate | Protected by adminMiddleware (constant-time admin key check). Only admin can toggle. |
| T-6-09 | Tampering | Optimistic UI | accept | Optimistic toggle reverts on API error. Worst case: brief UI flicker, no data corruption. |
| T-6-10 | Information Disclosure | Key table | mitigate | Raw token shown only on create (once). All GET /api/keys responses return masked tokens (server-side maskToken). |
| T-6-11 | Denial of Service | Upstream toggle | accept | Admin-only endpoint. No rate limit needed for admin actions at personal scale. |
| T-6-12 | Elevation of Privilege | /api/upstreams/:id/toggle | mitigate | Uses adminMiddleware, not authMiddleware. Access key holders cannot toggle upstreams. |
</threat_model>
