  <action>
    1. Create `internal/server/spa.go`:
       ```go
       package server

       import (
           "embed"
           "io/fs"
           "net/http"
           "strings"

           "github.com/gin-gonic/gin"
       )

       //go:embed all:web_dist
       var webFS embed.FS

       // spaHandler serves the embedded SPA files. For any path not matching a
       // static file, it falls back to index.html (SPA client-side routing).
       func spaHandler() gin.HandlerFunc {
           sub, err := fs.Sub(webFS, "web_dist")
           if err != nil {
               panic("embedded web_dist not found: " + err.Error())
           }
           fileServer := http.FileServer(http.FS(sub))

           return func(c *gin.Context) {
               path := c.Request.URL.Path

               // Skip API and relay routes — they are handled by registered handlers
               if strings.HasPrefix(path, "/api") || strings.HasPrefix(path, "/v1") || path == "/health" {
                   c.Next()
                   return
               }

               // Try to serve the static file
               f, err := sub.Open(strings.TrimPrefix(path, "/"))
               if err == nil {
                   f.Close()
                   fileServer.ServeHTTP(c.Writer, c.Request)
                   c.Abort()
                   return
               }

               // Fall back to index.html for SPA routing
               c.Request.URL.Path = "/"
               fileServer.ServeHTTP(c.Writer, c.Request)
               c.Abort()
           }
       }
       ```

       IMPORTANT: The `//go:embed` directive references `web_dist` (a directory that will be symlinked or copied during build). The Makefile handles placing `web/dist` content into `internal/server/web_dist/` before `go build`.

       ALTERNATIVE (simpler, preferred): Instead of symlink, use a build tag approach. Change the embed to reference a relative path. Since Go embed only works with paths relative to the source file, we need `web_dist/` to exist inside `internal/server/`. The Makefile copies it.

    2. Create `internal/server/web_dist/.gitkeep` as a placeholder so the directory exists for `go build` to work without a prior `make web-build`. Actually, better approach: use a build-tag so development builds work without the embed. Create TWO files:

       `internal/server/spa.go` (production, no build tag — always compiled):
       ```go
       package server

       import (
           "embed"
           "io/fs"
           "net/http"
           "strings"

           "github.com/gin-gonic/gin"
       )

       //go:embed all:web_dist
       var webFS embed.FS

       func spaHandler() gin.HandlerFunc {
           sub, err := fs.Sub(webFS, "web_dist")
           if err != nil {
               // web_dist is empty — portal not built, skip SPA serving
               return func(c *gin.Context) { c.Next() }
           }

           fileServer := http.FileServer(http.FS(sub))

           return func(c *gin.Context) {
               path := c.Request.URL.Path
               if strings.HasPrefix(path, "/api") || strings.HasPrefix(path, "/v1") || path == "/health" {
                   c.Next()
                   return
               }

               f, err := sub.Open(strings.TrimPrefix(path, "/"))
               if err == nil {
                   f.Close()
                   fileServer.ServeHTTP(c.Writer, c.Request)
                   c.Abort()
                   return
               }

               c.Request.URL.Path = "/"
               fileServer.ServeHTTP(c.Writer, c.Request)
               c.Abort()
           }
       }
       ```

       Create `internal/server/web_dist/.gitkeep` so the embed has something to embed even without a web build (Go 1.16+ requires the embedded directory to exist).
